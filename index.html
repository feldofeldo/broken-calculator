let selected=0, value=0, alpha=0, beta=0;

let history=[], states=[];

const $=id=>document.getElementById(id);

const fmt=n=>Number.isSafeInteger(n)?String(n):String(n);


function snapshot() {

 return {value,alpha,beta,history:history.slice()};

}

function renderTabs() {

 $("tabs").innerHTML=calculators.map((c,i)=>

  `<button class="tab ${i===selected?"active":""}" data-index="${i}">Calculator ${i+1}</button>`

 ).join("");

 $("tabs").querySelectorAll("button").forEach(b=>

  b.addEventListener("click",()=>choose(Number(b.dataset.index)))

 );

}

function choose(i) {

 selected=i;

 renderTabs();

 renderCalculator();

 reset();

}

function renderCalculator() {

 const c=calculators[selected];

 $("numberLabel").textContent="CALCULATOR "+(selected+1);

 $("calcTitle").textContent=c.title;

 $("rules").textContent=c.rules;

 $("memory").hidden=!c.memory;

 $("buttons").innerHTML=c.buttons.map(([name,label])=>

  `<button class="op" data-button="${name}">${name}<small>${label}</small></button>`

 ).join("");

 $("buttons").querySelectorAll("button").forEach(b=>

  b.addEventListener("click",()=>press(b.dataset.button))

 );

 $("challenges").innerHTML=c.challenges.map(q=>`<li>${q}</li>`).join("");

 $("teacherNotes").innerHTML=c.teacher;

 $("target").value=[20,102,99,2,55][selected];

}

function press(name) {

 states.push(snapshot());

 const old=value;

 let message="";

 if(selected===0) value+=name==="A"?3:7;

 else if(selected===1) value=name==="A"?value+1:value*3;

 else if(selected===2) {

  if(name==="A") value+=6;

  else if(name==="B") {

   if(value%2===0) value/=2;

   else message="No effect: the display is odd.";

  } else {

   if(value%3===0) value/=3;

   else message="No effect: the display is not divisible by 3.";

  }

 } else if(selected===3) {

  if(name==="A") value+=5;

  else if(name==="B") value+=7;

  else {

   const r=Math.sqrt(value);

   if(Number.isInteger(r)) value=r;

   else message="No effect: the display is not a perfect square.";

  }

 } else {

  const oldAlpha=alpha;

  if(name==="A") value+=alpha;

  else value+=alpha+beta;

  alpha=old;

  beta=oldAlpha;

 }

 history.push({name,from:old,to:value,message});

 $("status").textContent=message||`${name} pressed: ${fmt(old)} → ${fmt(value)}.`;

 render();

}

function render() {

 $("display").textContent=fmt(value);

 $("alpha").textContent=fmt(alpha);

 $("beta").textContent=fmt(beta);

 $("history").innerHTML=history.length?history.map((h,i)=>

  `<div class="step"><span>${i+1}. Button ${h.name}${h.message?" · no effect":""}</span><strong>${fmt(h.to)}</strong></div>`

 ).join(""):'<p class="muted">No buttons pressed yet.</p>';

 $("undo").disabled=history.length===0;

 $("targetResult").textContent="";

}

function reset() {

 value=calculators[selected].start;

 alpha=0; beta=0; history=[]; states=[];

 $("status").textContent="Ready! Press a button to begin.";

 render();

}

$("undo").addEventListener("click",()=>{

 if(!states.length)return;

 const s=states.pop();

 value=s.value; alpha=s.alpha; beta=s.beta; history=s.history;

 $("status").textContent="Last press undone.";

 render();

});

$("reset").addEventListener("click",reset);

$("copy").addEventListener("click",async()=>{

 const seq=history.map(h=>h.name).join("");

 try {

  await navigator.clipboard.writeText(seq||"(no presses)");

  $("status").textContent="Sequence copied: "+(seq||"(no presses)");

 } catch {

  $("status").textContent="Your sequence: "+(seq||"(no presses)");

 }

});

$("checkTarget").addEventListener("click",()=>{

 const target=Number($("target").value);

 if(!Number.isSafeInteger(target)) {

  $("targetResult").textContent="Enter a whole-number target.";

  return;

 }

 if(value===target) {

  $("targetResult").textContent=`Target reached in ${history.length} press${history.length===1?"":"es"}! Can you do it in fewer?`;

 } else {

  $("targetResult").textContent=`Current display: ${value}. Target: ${target}. Keep experimenting!`;

 }

});

renderTabs();

renderCalculator();

reset();

</script>
